        \item Well-defined. We address each of the conditions separately.
        \begin{itemize}
            \item \textbf{F1}: Obviously true. Suppose $x \in \N^2$. Thus we can write $x$ as $(m,n)$ where $m, n \in \N$. So their sum, $m + n$, must be a natural number and we can choose it to be $y$.
            \item \textbf{F2}: $x \in \N^2$. Thus we can write $x$ as $(m,n)$ where $m, n \in \N$. Suppose we have $y_1, y_2 \in \N$ such that $x \mathrel{g} y_1$ and $x \mathrel{g} y_2$. So $y_1 = m + n$ and $y_2 = m + n$ by definition of $g$, which immediately means $y_1 = y_2$.
        \end{itemize}
        Since both conditions are true, the function is well-defined.
        \item Ill-defined. Note that $1 \mathrel{h} (0,1)$ and $1 \mathrel{h} (1,0)$, so condition \textbf{F2} in \myref{definition:function} is not satisfied.
    \end{partquestions}

    % Composition of an Injection and Non-Injection
    \item \begin{partquestions}{\alph*}
        \item True. Suppose $h \circ f = h \circ g$. Thus $h(f(x)) = h(g(x))$ for all $x \in X$. As $h$ is injective, this means $f(x) = g(x)$ for all $x \in X$. Thus $f = g$.
        \item False. Let $X = \N$, and consider the functions
        \begin{align*}
            f &= \id_X,\\
            g &= \{(n, n) \vert n \text{ is even}\} \cup \{(n, 0) \vert n \text{ is odd}\}, \text{ and}\\
            h &= \{(n, 2n) \vert n \in \N\}.
        \end{align*}
        
        We first show that $h$ is injective. Suppose $n_1, n_2 \in \N$ such that $h(n_1) = h(n_2)$. So $2n_1 = 2n_2$ which clearly means $n_1 = n_2$. Thus $h$ is injective.

        Now we show $f \circ h = g \circ h$. We see that
        \begin{align*}
            (f \circ h)(n) &= f(h(n)) & (\text{Definition of composition})\\
            &= f(2n) & (\text{Definition of } h)\\
            &= 2n & (\text{Definition of } f)\\
            &= g(2n) & (\text{Definition of } g, \text{ since } 2n \text{ is even})\\
            &= g(h(n)) & (\text{Definition of } h)\\
            &= (g \circ h)(n) & (\text{Definition of composition})
        \end{align*}
        for any $n \in \N$, which thus means $f \circ h = g \circ h$.

        But one sees clearly that $(1, 1) \in f$ but $(1, 1) \notin g$, so $f \neq g$.
    \end{partquestions}

    % Composition of Surjective Functions is Surjective
    \item \begin{proof}
        Suppose $f: A \to B$ and $g: C \to D$ are surjective functions. Let $d \in D$. Since $g$ is surjective, there exists a $c \in C$ such that $g(c) = d$. In particular, since $g \circ f$ exists, we know that this $c$ is in the range of $f$, which means that $c \in B$. Since $f$ is surjective, we know that there exists an $a \in A$ such that $f(a) = c$ (remember, this $c$ is also in $B$). We note that this element $a \in A$ satisfies
        \[
            g(f(a)) = g(c) = d
        \]
        so a preimage of $d$ under $g \circ f$ is $a$. It follows that $g \circ f$ is surjective.
    \end{proof}
