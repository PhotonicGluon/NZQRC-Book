%=============== Counters ================
\newcounter{result}[section]
\renewcommand*{\theresult}{\arabic{chapter}.\arabic{section}.\arabic{result}}

%======== 'Results' Declarations =========
% Arguments to this command are:
% 1. Environment ID (e.g., theorem)
% 2. Environment name (e.g., Theorem)
% 3. Colour
% 4. Counter ID (e.g., result)
\newcommand{\makenewresultstyle}[4]{
    \newenvironment{#1}[1][]
    {
        \stepcounter{#4}
        \ifstrempty{##1}
        {
            \mdfsetup{
                frametitle={
                    \hspace{-3mm}
                    \tikz[baseline=(current bounding box.east),outer sep=0pt]
                    \node[anchor=east,rectangle,fill=#3]{\strut #2~\csname the#4\endcsname};
                }
            }
        }
        {
            \mdfsetup{
                frametitle={
                    \hspace{-3mm}
                    \tikz[baseline=(current bounding box.east),outer sep=0pt]
                    \node[anchor=east,rectangle,fill=#3]{\strut #2~\csname the#4\endcsname\textnormal{:~##1}};
                }
            }
        }
        \mdfsetup{
            innertopmargin=0pt,
            linecolor=#3,
            linewidth=2pt,
            frametitleaboveskip=\dimexpr-\ht\strutbox\relax
        }
        \begin{mdframed}[]\relax
    }
    {\end{mdframed}}
}

\makenewresultstyle{theorem}{Theorem}{Aquamarine3}{result}
\makenewresultstyle{lemma}{Lemma}{LightGoldenrod3}{result}
\makenewresultstyle{proposition}{Proposition}{DarkSeaGreen3}{result}
\makenewresultstyle{corollary}{Corollary}{NavajoWhite3}{result}
\makenewresultstyle{definition}{Definition}{SlateGray3}{result}
\makenewresultstyle{axiom}{Axiom}{Plum3}{result}

%====== Miscellaneous Declarations =======
% Example style
\newcommand*{\examplecolour}{DodgerBlue1}
\newenvironment{example}[1][]
{
    \stepcounter{result}
    \ifstrempty{#1}
    {
        \mdfsetup{
            frametitle={
                \hspace{-17.9px}
                \tikz[baseline=(current bounding box.east),outer sep=0pt]{
                    \draw[line width=2pt,\examplecolour](0,0)--(3mm,0);
                    \node at (3mm,0) [anchor=west,rectangle,line width=2pt,draw=\examplecolour,inner ysep=2.3pt]{\strut Example~\theresult};
                }
            }
        }
    }
    {
        \mdfsetup{
            frametitle={
                \hspace{-17.9px}
                \tikz[baseline=(current bounding box.east),outer sep=0pt]{
                    \draw[line width=2pt,\examplecolour](0,0)--(3mm,0);
                    \node at (3mm,0) [anchor=west,rectangle,line width=2pt,draw=\examplecolour,inner ysep=2.3pt]{\strut Example~\theresult\textnormal{:~#1}};
                }
            }
        }
    }
    \mdfsetup{
        innertopmargin=0pt,
        linecolor=\examplecolour,
        linewidth=2pt,
        frametitleaboveskip=\dimexpr-\ht\strutbox\relax,
        topline=false,
        rightline=false
    }
    \begin{mdframed}[]\relax
}
{\end{mdframed}}
