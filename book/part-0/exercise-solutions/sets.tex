    % Q4
    \item \begin{exercise}
        \begin{partquestions}{\alph*}
            \item Note that $S = \{-9, -7, -5, -3, -1, 1, 3, 5, 7, 9\}$, so $|S| = 10$.
            \item Let $A = \{1, 2, 3, 4, 5\}$ and $B = \{2, 3, 4, 5, 6, 7\}$. Thus $A \cup B = \{1, 2, 3, 4, 5, 6, 7\}$ and so $|A \cup B| = 7$. Hence one sees that $|A \cap B| = |\{2, 3, 4, 5\}| = 4$.
            \item $|A^p| = n^p$ since we are enumerating through the entirety of $A$ each of the $p$ times. Also $|\Power{A}| = 2^n$. The reason is an element of $A$ can either be in the subset or not, and that there are $n$ such elements.
        \end{partquestions}
