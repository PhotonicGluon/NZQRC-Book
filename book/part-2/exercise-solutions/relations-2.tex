    % Equivalence Relation Detective
    \item \begin{partquestions}{\alph*}
        \item Not an equivalence relation since $A$ is not reflexive (e.g., $1 \not\mathrel{A} 1$).
        
        \item Is an equivalence relation.
        \begin{itemize}
            \item \textbf{Reflexive}: We have $1 \mathrel{B} 1$, $2 \mathrel{B} 2$, and $3 \mathrel{B} 3$. Thus $B$ is reflexive.
            \item \textbf{Symmetric}: Suppose $x, y \in S$ where $x \mathrel{B} y$. We split into two cases.
            \begin{itemize}
                \item If $x = y$, reflexivity immediately implies $y \mathrel{B} x$.
                \item Otherwise symmetry is vacuously satisfied.
            \end{itemize}
            Thus $B$ is symmetric.
            \item \textbf{Transitive}: Suppose $x, y, z \in S$ where $x \mathrel{B} y$ and $y \mathrel{B} z$. We split into two cases.
            \begin{itemize}
                \item If $x = y = z$, we clearly have $x \mathrel{B} z$ by reflexivity.
                \item Otherwise transitivity is vacuously satisfied.
            \end{itemize}
            Thus $B$ is transitive.
        \end{itemize}
        Since $B$ is reflexive, symmetric, and transitive, it is an equivalence relation.

        \item Not an equivalence relation since it is not symmetric. In particular we have $1 \mathrel{C} 2$ but $2 \not\mathrel{C} 1$.

    % Equal Cardinality Relation
    \item \begin{partquestions}{\alph*}
        \item \begin{proof}
            We need to show that $\EqRel$ is reflexive, symmetric, and transitive.
            \begin{itemize}
                \item \textbf{Reflexive}: Let $A \in \Power{X}$. Clearly $|A| = |A|$ by using the identity function on $A$ (i.e., $\id_A$), so $A \EqRel A$, proving that $A$ is reflexive.
                
                \item \textbf{Symmetric}: Let $A, B \in \Power{X}$ such that $A \EqRel B$. This means $|A| = |B|$, which tells us that there exists a bijection $f: A \to B$. But $f^{-1}: B \to A$ is also a bijection, so $|B| = |A|$, i.e. $B \EqRel A$, proving that $\EqRel$ is symmetric.
                
                \item \textbf{Transitive}: Suppose $A, B, C \in \Power{X}$ such that $A \EqRel B$ and $B \EqRel C$. This means that $|A| = |B|$ and $|B| = |C|$, i.e. there exist bijections $f: A \to B$ and $g: B \to C$. But $g \circ f: A \to C$ is also a bijection by \myref{prop:composition-of-bijective-functions-is-bijective}, so $|A| = |C|$, i.e. $A \EqRel C$, proving that $\EqRel$ is transitive.
            \end{itemize}
            Thus we see that $\EqRel$ is an equivalence relation.
        \end{proof}

        \item Note that two sets have are in the same equivalence class if and only if they have the same cardinality. So the distinct equivalence classes of $\EqRel$ are
        \begin{itemize}
            \item $\{\emptyset\}$ (each set has cardinality 0);
            \item $\{\{1\}, \{2\}, \{3\}\}$ (each set has cardinality 1);
            \item $\{\{1, 2\}, \{1, 3\}, \{2, 3\}\}$ (each set has cardinality 2); and
            \item $\{\{1, 2, 3\}\}$ (each set has cardinality 3).
        \end{itemize}
    \end{partquestions}

    % Complete the Proof!
    \item \begin{proof}
        Suppose $\EqClass{x} \cap \EqClass{y} \neq \emptyset$. This means that there is an $a \in \EqClass{x} \cap \EqClass{y}$, i.e. $a \in \EqClass{x}$ and $a \in \EqClass{y}$. By definition of equivalence classes, this tells us that $a \EqRel x$ and $a \EqRel y$. We hence know that $x \EqRel a$ (by symmetry) and thus $x \EqRel y$ (by transitivity).
    \end{proof}

    % Symmetric and Antisymmetric?
    \item Yes. Consider $A = \{0\}$ and $R = \{(0, 0)\}$. Note that $R$ is symmetric since, for all $x$ (which must only be 0) and $y$ (which must also be 0), we see $x \mathrel{R} y$ means $y \mathrel{R} x$ (which is obvious as $0 \mathrel{R} 0$). We also see that $R$ is antisymmetric -- for all $x$ and $y$ (which are both 0), if $x\mathrel{R} y$ and $x \mathrel{R} y$ then $x = y$.
