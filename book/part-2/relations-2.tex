\section{Equivalence Relations}
Before we can define what an equivalence relation is, we define three properties that a relation could possess.

\begin{definition}[Reflexivity]
    A relation $R$ on a set $A$ is \term{reflexive}\index{relation!reflexive} if and only if
        \[
            \forall x \in A, x\mathrel{R}x.
        \]
\end{definition}

\begin{definition}[Symmetry]
    A relation $R$ on a set $A$ is \term{symmetric}\index{relation!symmetric} if and only if
    \[
        \forall x,y \in A, (x\mathrel{R}y \implies y\mathrel{R}x).
    \]
\end{definition}

\begin{definition}[Transitivity]
    A relation $R$ on a set $A$ is \term{transitive}\index{relation!transitive} if and only if
    \[
        \forall x,y,z \in A, (x\mathrel{R}y \land y\mathrel{R}z \implies x\mathrel{R}z).
    \]
\end{definition}

That was certainly a lot. Let us look at some examples to hammer in these definitions.

\begin{example}
    Let the set $A = \{0,1,2,3\}$. Define the relation
    \[
        R = \{(0,0),(0,1),(0,3),(1,0),(1,1),(2,2),(3,0),(3,3)\}. 
    \]
    It has the following directed graph.

    \begin{figure}[H]
        \centering
        \includegraphics[height=2.75cm]{images/part-2/relations-2/rst_1.png}
        \caption{Directed Graph of $R$}
    \end{figure}

    We note the following.
    \begin{itemize}
        \item $R$ is reflexive since $0\mathrel{R}0$, $1\mathrel{R}1$, and $2\mathrel{R}2$. This can be seen in the figure by the loops from each element to itself.
        \item $R$ is symmetric, since
        \begin{itemize}
            \item given $0\mathrel{R}1$ we also see $1\mathrel{R}0$; and
            \item given $0\mathrel{R}3$ we also see $3\mathrel{R}0$.
        \end{itemize}
        These are shown by the double arrows in the figure. For example, there is an arrow from 0 to 1 and an arrow from 1 to 0. There is also an arrow from 0 to 3 and an arrow from 3 to 0.
        \item $R$ is not transitive, since $1\mathrel{R}0$ and $0\mathrel{R}3$ but $1\not\mathrel{R}3$. This can be seen by the absence of an arrow from 1 to 3.
    \end{itemize}
\end{example}

\begin{example}
    Let the set $A = \{0,1,2,3\}$. Define the relation
    \[
        S = \{(0,0),(0,2),(0,3),(2,3)\}
    \]
    on $A$, which has the following directed graph.

    \begin{figure}[H]
        \centering
        \includegraphics[height=2.75cm]{images/part-2/relations-2/rst_2.png}
        \caption{Directed Graph of $S$}
    \end{figure}

    We note the following.
    \begin{itemize}
        \item $S$ is not reflexive since $1\not\mathrel{S}1$. This can be seen by the absence of a loop from 1 to 1.
        \item $S$ is not symmetric since, while $0\mathrel{S}2$, we see $2\not\mathrel{S}0$. This can be seen in the diagram by an arrow from 0 to 2 but the absence of an arrow from 2 to 0.
        \item $S$ is transitive since $0\mathrel{S}2$ and $2\mathrel{S}3$ means $0\mathrel{S}3$. This is shown in the diagram by an arrow from 0 to 3 after there were arrows from 0 to 2 and 2 to 3.
    \end{itemize}
\end{example}

With these three definitions, we can now define equivalence relations.

\begin{definition}[Equivalence Relation]
    An \term{equivalence relation on a set $A$}\index{equivalence relation}, usually denoted $\eqrel$, is a relation on $A$ that is reflexive, symmetric, and transitive.
\end{definition}

\begin{example}[An Equivalence Relation]
    Consider $A = \{0, 1, 2, 3\}$ and define a relation $\eqrel$ on $A$ by
    \[
        \eqrel\;= \{(0,0), (1,1), (2,2), (3,3)\}
    \]
    which has the following arrow diagram.
    \begin{figure}[H]
        \centering
        \includegraphics[height=2.75cm]{images/part-2/relations-2/equiv-rel_1.png}
        \caption{Directed Graph of $\eqrel$}
    \end{figure}

    We note that $\eqrel$ is
    \begin{itemize}
        \item reflexive since $0\mathrel{\eqrel}0$, $1\mathrel{\eqrel}1$, $2\mathrel{\eqrel}2$, and $3\mathrel{\eqrel}3$;
        \item symmetric since the symmetric requirements are implied by reflexivity; and
        \item transitive by vacuous definition of transitivity.
    \end{itemize}
    Therefore $\eqrel$ is an equivalence relation.
\end{example}

\begin{example}[Another Equivalence Relation]
    Consider $A = \{0, 1, 2, 3\}$ and define a relation $\eqrel$ on $A$ by
    \begin{align*}
        \eqrel &= \{(0, 0), (0, 1), (0, 2), (1, 0), (1, 1),\\
        &\quad\quad(1, 2), (2, 0), (2, 1), (2, 2), (3, 3)\}
    \end{align*}
    which has the following arrow diagram.
    \begin{figure}[H]
        \centering
        \includegraphics[height=2.75cm]{images/part-2/relations-2/equiv-rel_2.png}
        \caption{Directed Graph of $\eqrel$}
    \end{figure}

    We note that $\eqrel$ is
    \begin{itemize}
        \item reflexive since $0\mathrel{\eqrel}0$, $1\mathrel{\eqrel}1$, $2\mathrel{\eqrel}2$, and $3\mathrel{\eqrel}3$;
        \item symmetric since
        \begin{itemize}
            \item given $0\mathrel{\eqrel}1$ we have $1\mathrel{\eqrel}0$;
            \item given $0\mathrel{\eqrel}2$ we have $2\mathrel{\eqrel}0$; and
            \item given $1\mathrel{\eqrel}2$ we have $2\mathrel{\eqrel}1$.
        \end{itemize}
        All other symmetric requirements are implied by reflexivity; and
        \item transitive since
        \begin{itemize}
            \item given $0\mathrel{\eqrel}1$ and $1\mathrel{\eqrel}2$ we have $0\mathrel{\eqrel}2$;
            \item given $1\mathrel{\eqrel}2$ and $2\mathrel{\eqrel}0$ we have $1\mathrel{\eqrel}0$; and
            \item given $2\mathrel{\eqrel}0$ and $0\mathrel{\eqrel}1$ we have $2\mathrel{\eqrel}1$.
        \end{itemize}
        All other transitive requirements are implied by prior properties.
    \end{itemize}
    Therefore $\eqrel$ is an equivalence relation.
\end{example}

\begin{example}[An Equivalence Relation on a Formal Language]
    \setlength{\parindent}{1.5em}

    Let $\Alphabet$ be the set of English letters, i.e. $\Alphabet = \{\text{a}, \text{b}, \text{c}, \dots, \text{x}, \text{y}, \text{z}\}$. Let $\Lang$ be the formal language using the alphabet $\Alphabet$ such that it contains strings formed from $\Alphabet$.

    Define a function $\mathrm{Start}_n: \Lang \to \Lang$, where $n \in \N \setminus \{0\}$, such that
    \[
        \mathrm{Start}_n(s) = \begin{cases}
            s & \text{if length of } s \text{ is less than } n\\
            \text{the first } n \text{ characters of } s & \text{otherwise}
        \end{cases}
    \]
    for all $s \in \Lang$. For example, $\mathrm{Start}_1(\text{abc}) = \text{a}$, $\mathrm{Start}_2(\text{abc}) = \text{ab}$, $\mathrm{Start}_3(\text{abc}) = \text{abc}$, and $\mathrm{Start}_4(\text{abc}) = \text{abc}$.

    We can now define a relation $\eqrel$ on $\Lang$ by
    \[
        s\mathrel{\eqrel}t \iff \mathrm{Start}_3(s) = \mathrm{Start}_3(t)
    \]
    for all $s, t \in \Lang$. We show that $\eqrel$ is an equivalence relation.
    \begin{proof}
        We need to show that $\eqrel$ is reflexive, symmetric, and transitive.
        \begin{itemize}
            \item \textbf{Reflexive}: Let $s \in \Lang$. Clearly $s$ has the same first 3 letters as itself, i.e. $\mathrm{Start}_3(s) = \mathrm{Start}_3(s)$. Thus $s \mathrel{\eqrel} s$, proving that $s$ is reflexive.
            \item \textbf{Symmetric}: Let $s, t \in \Lang$ such that $s \mathrel{\eqrel} t$. This means that $\mathrm{Start}_3(s) = \mathrm{Start}_3(t)$. But clearly this means that $\mathrm{Start}_3(t) = \mathrm{Start}_3(s)$, i.e. $t \mathrel{\eqrel} s$, proving that $\eqrel$ is symmetric.
            \item \textbf{Transitive}: Let $s, t, u \in \Lang$ such that $s \mathrel{\eqrel} t$ and $t \mathrel{\eqrel} u$. So we have $\mathrm{Start}_3(s) = \mathrm{Start}_3(t)$ and $\mathrm{Start}_3(t) = \mathrm{Start}_3(u)$. But clearly this means that $\mathrm{Start}_3(s) = \mathrm{Start}_3(u)$, i.e. $s \mathrel{\eqrel} u$, proving that $\eqrel$ is transitive.
        \end{itemize}
        Thus we see that $\eqrel$ is an equivalence relation.
    \end{proof}
\end{example}

        \item $D = \{(1, 1), (1, 2), (2, 1), (2, 2), (2, 3), (3, 2), (3, 3)\}$
    \end{partquestions}
\end{exercise}
