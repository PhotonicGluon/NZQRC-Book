%======== 'Results' Environments =========
% Arguments to this command are:
% 1. Environment ID (e.g., theorem)
% 2. Environment name (e.g., Theorem)
% 3. Colour
% 4. Counter ID (e.g., result)
\newcommand{\makenewresultstyle}[4]{
    \newaliascnt{#1}{#4}
    \newenvironment{#1}[1][]
    {
        \refstepcounter{#1}
        \ifstrempty{##1}
        {
            \mdfsetup{
                frametitle={
                    \hspace{-3mm}
                    \tikz[baseline=(current bounding box.east),outer sep=0pt]
                    \node[anchor=east,rectangle,fill=#3]{\strut #2~\csname the#4\endcsname};
                }
            }
        }
        {
            \mdfsetup{
                frametitle={
                    \hspace{-3mm}
                    \tikz[baseline=(current bounding box.east),outer sep=0pt]
                    \node[anchor=east,rectangle,fill=#3]{\strut #2~\csname the#4\endcsname\textnormal{:~##1}};
                }
            }
        }
        \mdfsetup{
            innertopmargin=0pt,
            linecolor=#3,
            linewidth=2pt,
            frametitleaboveskip=\dimexpr-\ht\strutbox\relax
        }
        \begin{mdframed}[]\relax
    }
    {\end{mdframed}}
}

\makenewresultstyle{theorem}{Theorem}{Aquamarine3}{result}

\makenewresultstyle{corollary}{Corollary}{NavajoWhite3}{result}

\makenewresultstyle{axiom}{Axiom}{Plum3}{result}

%====== Miscellaneous Environments =======
% Arguments to this command are:
% 1. Counter ID, if provided (e.g., result)
% 2. Environment ID (e.g., theorem)
% 3. Environment name (e.g., Theorem)
% 4. Colour
\newcommand{\makenewmiscstyle}[4][]{
    \ifstrempty{#1}{}{\newaliascnt{#2}{#1}}
    \newenvironment{#2}[1][]
    {
        \ifstrempty{#1}{}{\refstepcounter{#2}}
        \ifstrempty{##1}
        {
            \mdfsetup{
                frametitle={
                    \hspace{-17.9px}
                    \tikz[baseline=(current bounding box.east),outer sep=0pt]{
                        \draw[line width=2pt,#4](0,0)--(3mm,0);
                        \node at (3mm,0) [anchor=west,rectangle,line width=2pt,draw=#4,inner ysep=2.3pt]{
                            \strut #3\ifstrempty{#1}{}{~\csname the#1\endcsname}
                        };
                    }
                }
            }
        }
        {
            \mdfsetup{
                frametitle={
                    \hspace{-17.9px}
                    \tikz[baseline=(current bounding box.east),outer sep=0pt]{
                        \draw[line width=2pt,#4](0,0)--(3mm,0);
                        \node at (3mm,0) [anchor=west,rectangle,line width=2pt,draw=#4,inner ysep=2.3pt]{
                            \strut #3\ifstrempty{#1}{}{~\csname the#1\endcsname}\textnormal{:~##1}
                        };
                    }
                }
            }
        }
        \mdfsetup{
            innertopmargin=0pt,
            linecolor=#4,
            linewidth=2pt,
            frametitleaboveskip=\dimexpr-\ht\strutbox\relax,
            topline=false,
            rightline=false
        }
        \begin{mdframed}[]\relax
    }
    {\end{mdframed}}
}
