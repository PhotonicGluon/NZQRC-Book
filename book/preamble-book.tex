\makeatother

\newcommand*{\Lettrine}[2][]{%
    % Set default key values
    \def\thislines{2}%
    \def\thisdepth{0}%
    \def\thisloversize{0}%
    \def\thislraise{0}%
    \def\thisfindent{0pt}%
    \def\thisnindent{0.25em}%
    % Set keys
    \setkeys{lettrine-key}{#1}%
    % Get appropriate strings
    \StrRemoveBraces{#2}[\LettrineStr]%
    \StrChar{\LettrineStr}{1}[\FirstChar]%
    \StrMid{\LettrineStr}{2}{1000}[\RemainingChars]%
    % Format lettrine
    \lettrine[%
        lines=\thislines,%
        depth=\thisdepth,%
        loversize=\thisloversize,%
        lraise=\thislraise,%
        findent=\thisfindent,%
        nindent=\thisnindent%
    ]{\FirstChar}{}\RemainingChars
}

% Arguments for this command are:
% 1. Part title
% 2. Head shot image for the post-part page content
% 3. Post-part page quote
% 4. Quote author and date
% 5. Quote Citation
\newcommand{\Part}[5]{
    % Set up
    \cleardoubleoddpage
    \color{white}
    
    % First page, containing the part title
    \part{#1}

    % Second page, containing the quote
    \thispagestyle{partpage}
    \vspace*{\fill}
    \vspace{-2cm}
    \begin{center}
        \Ifstr{#2}{}{
            % No head shot to put
        }{
            \begin{figure}[H]
                \centering
                \includegraphics[height=4cm]{#2}
            \end{figure}
        }
        \hrulefill[0.75pt]
        \raisebox{-5mm}{
            \hspace{2mm}
            \includegraphics[height=1cm]{images/part-\number\value{part}/preamble/qr.png}
            \hspace{2mm}
        }
        \hrulefill[0.75pt]
        \quoteattr[0.5cm]{#3}{#4}{#5}
        \hrule height 2pt
    \end{center}
    \vspace*{\fill}

    % Clean up
    \cleardoubleemptypage
    \color{black}
    \pagecolor{white}
}
