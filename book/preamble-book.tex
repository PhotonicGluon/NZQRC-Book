\newcommand*{\Lettrine}[2][]{%
    % Set default key values
    \def\thislines{2}%
    \def\thisdepth{0}%
    \def\thisfindent{0pt}%
    \def\thisnindent{0.25em}%
    % Set keys
    \setkeys{lettrine-key}{#1}%
    % Get appropriate strings
    \StrRemoveBraces{#2}[\LettrineStr]%
    \StrChar{\LettrineStr}{1}[\FirstChar]%
    \StrMid{\LettrineStr}{2}{1000}[\RemainingChars]%
    % Format lettrine
    \lettrine[%
        lines=\thislines,%
        depth=\thisdepth,%
        findent=\thisfindent,%
        nindent=\thisnindent,%
        realheight%
    ]{\FirstChar}{}\RemainingChars
}
